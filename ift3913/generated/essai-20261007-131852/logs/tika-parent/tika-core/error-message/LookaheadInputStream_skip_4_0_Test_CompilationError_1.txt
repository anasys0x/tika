package org.apache.tika.io;

import org.apache.tika.io.LookaheadInputStream;
import java.io.ByteArrayInputStream;
import java.io.IOException;
import java.lang.reflect.Method;
import org.junit.jupiter.api.extension.ExtendWith;
import org.mockito.junit.jupiter.MockitoExtension;
import java.io.InputStream;
import org.mockito.*;
import org.junit.jupiter.api.*;
import static org.mockito.Mockito.*;
import static org.junit.jupiter.api.Assertions.*;

public class LookaheadInputStream_skip_4_0_Test {

    private LookaheadInputStream lookaheadInputStream;

    private ByteArrayInputStream inputStream;

    @BeforeEach
    public void setUp() {
        inputStream = new ByteArrayInputStream(new byte[] { 1, 2, 3, 4, 5 });
        lookaheadInputStream = new LookaheadInputStream(inputStream, 5);
    }

    @Test
    public void testSkip() throws IOException, NoSuchMethodException, IllegalAccessException, java.lang.reflect.InvocationTargetException {
        Method skipMethod = LookaheadInputStream.class.getDeclaredMethod("skip", long.class);
        skipMethod.setAccessible(true);
        // Test skipping less than available bytes
        long result = (long) skipMethod.invoke(lookaheadInputStream, 2);
        assertEquals(2, result);
        assertEquals(2, lookaheadInputStream.getPosition());
        // Test skipping more than available bytes
        result = (long) skipMethod.invoke(lookaheadInputStream, 10);
        assertEquals(3, result);
        assertEquals(5, lookaheadInputStream.getPosition());
        // Test skipping zero bytes
        result = (long) skipMethod.invoke(lookaheadInputStream, 0);
        assertEquals(0, result);
        assertEquals(5, lookaheadInputStream.getPosition());
        // Test skipping after reaching end of stream
        result = (long) skipMethod.invoke(lookaheadInputStream, 1);
        assertEquals(0, result);
        assertEquals(5, lookaheadInputStream.getPosition());
    }

    @Test
    public void testSkipWithEmptyStream() throws IOException, NoSuchMethodException, IllegalAccessException, java.lang.reflect.InvocationTargetException {
        inputStream = new ByteArrayInputStream(new byte[0]);
        lookaheadInputStream = new LookaheadInputStream(inputStream, 5);
        Method skipMethod = LookaheadInputStream.class.getDeclaredMethod("skip", long.class);
        skipMethod.setAccessible(true);
        long result = (long) skipMethod.invoke(lookaheadInputStream, 1);
        assertEquals(0, result);
        assertEquals(0, lookaheadInputStream.getPosition());
    }

    @Test
    public void testSkipWithNullStream() throws IOException, NoSuchMethodException, IllegalAccessException, java.lang.reflect.InvocationTargetException {
        lookaheadInputStream = new LookaheadInputStream(null, 5);
        Method skipMethod = LookaheadInputStream.class.getDeclaredMethod("skip", long.class);
        skipMethod.setAccessible(true);
        long result = (long) skipMethod.invoke(lookaheadInputStream, 1);
        assertEquals(0, result);
        assertEquals(0, lookaheadInputStream.getPosition());
    }
}
