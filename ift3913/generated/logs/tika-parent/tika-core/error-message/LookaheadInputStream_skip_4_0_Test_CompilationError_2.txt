import org.apache.tika.io.LookaheadInputStream;

import org.junit.jupiter.api.extension.ExtendWith;
import org.mockito.junit.jupiter.MockitoExtension;
import java.io.InputStream;
import org.mockito.*;
import org.junit.jupiter.api.*;
import static org.mockito.Mockito.*;
import static org.junit.jupiter.api.Assertions.*;

@ExtendWith(MockitoExtension.class)
public class LookaheadInputStream_skip_4_0_Test {

    private LookaheadInputStream lookaheadInputStream;

    private ByteArrayInputStream inputStream;

    @BeforeEach
    public void setUp() {
        byte[] data = "Hello, World!".getBytes();
        inputStream = new ByteArrayInputStream(data);
        lookaheadInputStream = new LookaheadInputStream(inputStream, 1024);
    }

    @Test
    public void testSkip() throws IOException, NoSuchMethodException, IllegalAccessException, java.lang.reflect.InvocationTargetException {
        Method skipMethod = LookaheadInputStream.class.getDeclaredMethod("skip", long.class);
        skipMethod.setAccessible(true);
        // Test skipping less than available bytes
        long skipped = (long) skipMethod.invoke(lookaheadInputStream, 5);
        assertEquals(5, skipped);
        assertEquals(5, lookaheadInputStream.getPosition());
        // Test skipping more than available bytes
        skipped = (long) skipMethod.invoke(lookaheadInputStream, 100);
        assertEquals(6, skipped);
        assertEquals(11, lookaheadInputStream.getPosition());
        // Test skipping when no more bytes are available
        skipped = (long) skipMethod.invoke(lookaheadInputStream, 100);
        assertEquals(0, skipped);
        assertEquals(11, lookaheadInputStream.getPosition());
    }

    @Test
    public void testSkipWithIOException() throws IOException, NoSuchMethodException, IllegalAccessException, java.lang.reflect.InvocationTargetException {
        Method skipMethod = LookaheadInputStream.class.getDeclaredMethod("skip", long.class);
        skipMethod.setAccessible(true);
        // Close the underlying stream to simulate an IOException
        inputStream.close();
        assertThrows(IOException.class, () -> {
            skipMethod.invoke(lookaheadInputStream, 5);
        });
    }
}
